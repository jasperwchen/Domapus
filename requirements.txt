# What the data run installs. Test and lint tools are in requirements-dev.txt.
#
# Pinned exactly: the pipeline's numbers depend on library semantics, so an unpinned range
# lets a release change published data with no diff to review.
#
# Every pin must have a cp314 wheel (`.python-version` is 3.14). Installs use
# `--only-binary=:all:` so a missing wheel fails at once instead of compiling Arrow C++ for
# four minutes into an unrelated CMake error. Check with `pip index versions <pkg>`.

# Newest 2.x with cp314 wheels. 3.x (mandatory copy-on-write) is a deliberate migration.
pandas==2.3.3
# 2.33.0 patches GHSA-gc5v-m9x4-r6x2.
requests==2.33.0
# CSV streaming ingest; typed columns keep "07002" a string. 25.x is past GHSA-rgxp-2hwp-jwgg.
pyarrow==25.0.1
# `noise.py` fits K on the panel every run, and every reliability tier comes from that fit.
numpy==2.5.2
# cKDTree for `spatial.py`. libpysal and esda are deliberately refused.
scipy==1.18.1
