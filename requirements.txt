# Pinned exactly. The pipeline's correctness depends on pandas semantics, so an
# unpinned range means a library release can silently change published numbers
# with no diff to review. This file was `pandas>=1.5.0` while PyPI's current
# release was 3.0.5 — the next scheduled run would have installed a major
# version with mandatory copy-on-write straight into the sort/dedup path.
#
# pandas 2.3.3 is the newest 2.x with cp314 wheels (2.3.2 and older have none,
# and CI pins Python 3.14, so anything older would build from source).
# Moving to pandas 3.x is a deliberate migration, not an accident: it makes
# copy-on-write mandatory and changes several dtype behaviours.
pandas==2.3.3
requests==2.33.0
pytest==9.0.3

# pyarrow does the CSV streaming ingest: 1.33 GB in ~35 s against pandas'
# minutes, and its typed `include_columns` is what keeps "07002" a string
# rather than the integer 7002.
pyarrow==23.0.1
numpy==2.2.4
